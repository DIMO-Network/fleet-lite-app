```go
package service

import (
	"context"
	"encoding/json"
	"fmt"
	"math"
	"net/http"
	"sort"
	"time"

	"github.com/DIMO-Network/parentos/internal/models"
	"github.com/patrickmn/go-cache"
	"github.com/rs/zerolog"
)

const tripSummaryCacheTTL = 45 * time.Second
const tripSummaryWindow = 14 * 24 * time.Hour

// TripSummary is the computed post-trip safety report for one vehicle.
type TripSummary struct {
	StartedAt       string  `json:"startedAt"`
	EndedAt         string  `json:"endedAt"`
	DurationSeconds int64   `json:"durationSeconds"`
	DistanceMiles   float64 `json:"distanceMiles"`
	TopSpeedMph     float64 `json:"topSpeedMph"`
	StartAddress    string  `json:"startAddress"`
	EndAddress      string  `json:"endAddress"`
	StartLat        float64 `json:"startLat"`
	StartLon        float64 `json:"startLon"`
	EndLat          float64 `json:"endLat"`
	EndLon          float64 `json:"endLon"`
}

type TripSummaryService struct {
	telemetry  TelemetryAPIService
	cache      *cache.Cache
	httpClient *http.Client
	logger     zerolog.Logger
}

func NewTripSummaryService(telemetry TelemetryAPIService, logger zerolog.Logger) *TripSummaryService {
	return &TripSummaryService{
		telemetry:  telemetry,
		cache:      cache.New(tripSummaryCacheTTL, 2*tripSummaryCacheTTL),
		httpClient: &http.Client{Timeout: 5 * time.Second},
		logger:     logger,
	}
}

// LastTripSummary returns the most recent completed trip for tokenID, or nil
// if no trips exist in the last 14 days. Results are cached for 45 seconds.
func (s *TripSummaryService) LastTripSummary(ctx context.Context, tenant models.Tenant, tokenID uint64) (*TripSummary, error) {
	cacheKey := fmt.Sprintf("%s:%d", tenant.ID, tokenID)
	if v, ok := s.cache.Get(cacheKey); ok {
		if v == nil {
			return nil, nil
		}
		return v.(*TripSummary), nil
	}

	now := time.Now().UTC()
	from := now.Add(-tripSummaryWindow).Format(time.RFC3339)
	to := now.Format(time.RFC3339)

	// Detect mechanism: aftermarket → frequencyAnalysis, else ignitionDetection.
	// TelemetryAPIService.Segments handles the auto-fallback internally.
	segments, err := s.telemetry.Segments(tenant, tokenID, from, to, "ignitionDetection")
	if err != nil || len(segments) == 0 {
		if err == nil {
			// Try frequency-based fallback before giving up.
			segments, err = s.telemetry.Segments(tenant, tokenID, from, to, "frequencyAnalysis")
		}
		if err != nil || len(segments) == 0 {
			s.cache.Set(cacheKey, nil, tripSummaryCacheTTL)
			return nil, nil
		}
	}

	// Newest first — find the most recent completed (non-ongoing) trip.
	sort.Slice(segments, func(i, j int) bool {
		return segments[i].Start.Timestamp > segments[j].Start.Timestamp
	})
	var trip *Segment
	for i := range segments {
		if !segments[i].IsOngoing {
			trip = &segments[i]
			break
		}
	}
	if trip == nil {
		s.cache.Set(cacheKey, nil, tripSummaryCacheTTL)
		return nil, nil
	}

	startT, _ := time.Parse(time.RFC3339, trip.Start.Timestamp)
	endT, _ := time.Parse(time.RFC3339, trip.End.Timestamp)
	durationSec := int64(endT.Sub(startT).Seconds())

	// Fetch route points for distance.
	routePts, err := s.telemetry.RoutePoints(tenant, tokenID, trip.Start.Timestamp, trip.End.Timestamp)
	distanceMiles := 0.0
	if err == nil && len(routePts) > 1 {
		var km float64
		for i := 1; i < len(routePts); i++ {
			km += haversineKm(routePts[i-1].Lat, routePts[i-1].Lon, routePts[i].Lat, routePts[i].Lon)
		}
		distanceMiles = km * 0.621371
	}

	// Fetch max speed (km/h → mph) via 5-minute buckets; take the max across
	// all buckets. Using a fixed "5m" interval avoids guessing the telemetry
	// API's duration format and gives enough granularity for peak detection.
	topSpeedMph := 0.0
	buckets, err := s.telemetry.TimeSeries(tenant, tokenID, "speed",
		trip.Start.Timestamp, trip.End.Timestamp, "5m")
	if err == nil {
		for _, b := range buckets {
			if mph := b.Max * 0.621371; mph > topSpeedMph {
				topSpeedMph = mph
			}
		}
	}

	summary := &TripSummary{
		StartedAt:       trip.Start.Timestamp,
		EndedAt:         trip.End.Timestamp,
		DurationSeconds: durationSec,
		DistanceMiles:   math.Round(distanceMiles*10) / 10,
		TopSpeedMph:     math.Round(topSpeedMph*10) / 10,
		StartLat:        trip.Start.Value.Latitude,
		StartLon:        trip.Start.Value.Longitude,
		EndLat:          trip.End.Value.Latitude,
		EndLon:          trip.End.Value.Longitude,
	}

	// Best-effort reverse geocode — never fail the whole request if it errors.
	summary.StartAddress = s.reverseGeocode(trip.Start.Value.Latitude, trip.Start.Value.Longitude)
	summary.EndAddress = s.reverseGeocode(trip.End.Value.Latitude, trip.End.Value.Longitude)

	s.cache.Set(cacheKey, summary, tripSummaryCacheTTL)
	return summary, nil
}

// reverseGeocode calls OSM Nominatim. Returns "lat, lon" string on failure.
func (s *TripSummaryService) reverseGeocode(lat, lon float64) string {
	url := fmt.Sprintf("https://nominatim.openstreetmap.org/reverse?lat=%f&lon=%f&format=json", lat, lon)
	req, err := http.NewRequest("GET", url, nil)
	if err != nil {
		return fmt.Sprintf("%.4f, %.4f", lat, lon)
	}
	req.Header.Set("User-Agent", "parentos/1.0 (contact@parentos.app)")
	resp, err := s.httpClient.Do(req)
	if err != nil {
		return fmt.Sprintf("%.4f, %.4f", lat, lon)
	}
	defer resp.Body.Close()
	var result struct {
		DisplayName string `json:"display_name"`
	}
	if err := json.NewDecoder(resp.Body).Decode(&result); err != nil || result.DisplayName == "" {
		return fmt.Sprintf("%.4f, %.4f", lat, lon)
	}
	return result.DisplayName
}

func haversineKm(lat1, lon1, lat2, lon2 float64) float64 {
	const R = 6371.0
	dLat := (lat2 - lat1) * math.Pi / 180
	dLon := (lon2 - lon1) * math.Pi / 180
	a := math.Sin(dLat/2)*math.Sin(dLat/2) +
		math.Cos(lat1*math.Pi/180)*math.Cos(lat2*math.Pi/180)*
			math.Sin(dLon/2)*math.Sin(dLon/2)
	return 2 * R * math.Asin(math.Sqrt(a))
}
```

```go
package controllers

import (
	"github.com/DIMO-Network/parentos/internal/service"
	"github.com/gofiber/fiber/v2"
	"github.com/rs/zerolog"
)

type TripSummaryController struct {
	logger     *zerolog.Logger
	vehicleSvc *service.VehicleService
	summarySvc *service.TripSummaryService
}

func NewTripSummaryController(logger *zerolog.Logger, vehicleSvc *service.VehicleService, summarySvc *service.TripSummaryService) *TripSummaryController {
	return &TripSummaryController{logger: logger, vehicleSvc: vehicleSvc, summarySvc: summarySvc}
}

// GetLastTripSummary — GET /telemetry/:tokenID/last-trip-summary
func (t *TripSummaryController) GetLastTripSummary(c *fiber.Ctx) error {
	tenant, err := GetTenant(c)
	if err != nil {
		return fiber.NewError(fiber.StatusBadRequest, err.Error())
	}
	tokenID, err := ParseTokenIDParam(c, "tokenID")
	if err != nil {
		return err
	}
	if _, verr := t.vehicleSvc.GetVehicle(c.Context(), tenant.ID, int64(tokenID)); verr != nil {
		return fiber.NewError(fiber.StatusForbidden, "vehicle is not part of this family")
	}
	summary, err := t.summarySvc.LastTripSummary(c.Context(), tenant, tokenID)
	if err != nil {
		t.logger.Err(err).Uint64("tokenID", tokenID).Msg("trip summary failed")
		return fiber.NewError(fiber.StatusBadGateway, "trip summary unavailable")
	}
	return c.JSON(fiber.Map{"summary": summary})
}
```

```typescript
/** Format duration in seconds to "Xh Ym" or "Xm" string. */
export function formatTripDuration(seconds: number): string {
    const h = Math.floor(seconds / 3600);
    const m = Math.floor((seconds % 3600) / 60);
    if (h > 0) return `${h}h ${m}m`;
    return `${m}m`;
}

/** Format an ISO timestamp to "Jun 17 · 10:42pm" */
export function formatTripDate(iso: string): string {
    const d = new Date(iso);
    return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' }) +
        ' · ' + d.toLocaleTimeString('en-US', { hour: 'numeric', minute: '2-digit', hour12: true }).toLowerCase();
}

/** Speed class: 'ok' | 'warn' | 'danger' */
export function speedClass(mph: number): string {
    if (mph > 85) return 'danger';
    if (mph > 75) return 'warn';
    return 'ok';
}
```
